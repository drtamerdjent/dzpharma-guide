    75→function getContributorHandle(): string {
    76→  if (typeof window === "undefined") return "Contributeur anonyme";
    77→  try {
    78→    let handle = window.localStorage.getItem(HANDLE_KEY);
    79→    if (!handle) {
    80→      handle = `Contributeur #${Math.floor(1000 + Math.random() * 9000)}`;
    81→      window.localStorage.setItem(HANDLE_KEY, handle);
    82→    }
    83→    return handle;
    84→  } catch {
    85→    return "Contributeur anonyme";
    86→  }
    87→}
